    <script>
        // 1. 初始化 NiiVue
        const nv = new niivue.Niivue({
            onImageLoaded: () => { console.log("影像載入成功！"); },
            loadingText: "正在努力載入大腦中...",
            backColor: [0, 0, 0, 1],
            show3Dcrosshair: true,
            isColorbar: true // 顯示右側數值色標
        });

        nv.attachTo('canvas');

        // 使用 try-catch 來抓取隱藏錯誤
        async function loadBrain() {
            try {
                const volumeList = [
                    { url: "mni152.nii.gz", colorMap: "gray" },
                    { url: "stats_map.nii.gz", colorMap: "redyell", cal_min: 3.09, cal_max: 8.0 }
                ];
                await nv.loadVolumes(volumeList);
                console.log("NiiVue 載入完成");
            } catch (err) {
                // 如果失敗，直接在網頁畫面上顯示錯誤字樣
                document.body.innerHTML += `<div style="color:red; position:absolute; top:50%; width:100%; text-align:center;">載入失敗原因: ${err}</div>`;
            }
        }

loadBrain();

        nv.attachTo('canvas');
        
        // 設定顯示模式為 3D 渲染 (最吸睛)
        nv.setSliceType(nv.sliceTypeRender);

        // 2. 載入資料
        const volumeList = [
            {
                url: "icbm_mni152_t1_09a_asym_2.nii.gz",
                colorMap: "gray",
                opacity: 1,
                visible: true
            },
            {
                url: "spmF_0004_5.nii",
                colorMap: "redyell", // 使用紅-黃漸層
                cal_min: 3.09,       // p < 0.001 的門檻 (假設是 z-score)
                cal_max: 8.0,        // 最高亮度數值
                opacity: 1,
                visible: true
            }
        ];

        nv.loadVolumes(volumeList);

        // 3. 自動旋轉邏輯
        let azimuth = 0;
        let rotateInterval;
        let resetTimer;

        function startRotation() {
            rotateInterval = setInterval(() => {
                azimuth = (azimuth + 0.5) % 360; // 調整 0.5 改速度
                nv.setAzimuth(azimuth);
            }, 30);
        }

        function stopRotation() {
            clearInterval(rotateInterval);
            clearTimeout(resetTimer);
            // 10秒無人操作後重新開始旋轉
            resetTimer = setTimeout(startRotation, 10000);
        }

        // 4. 監聽互動事件 (有人操作時暫停旋轉)
        const canvas = document.getElementById('canvas');
        canvas.addEventListener('touchstart', stopRotation);
        canvas.addEventListener('mousedown', stopRotation);
        canvas.addEventListener('wheel', stopRotation);

        // 啟動初始旋轉
        startRotation();

    </script>
